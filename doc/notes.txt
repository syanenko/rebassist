// Assume an SVG path element with id "curve1" exists
// Example path d attribute: "M 48 84 Q 100 187 166 37"
let curveEl = document.getElementById('curve1');
let curve = Bezier.fromSVG(curveEl.getAttribute('d')); //

// Array to store the line points
let linePoints = [];
const numSegments = 10; // More segments = smoother curve

// Sample points along the curve
for (let i = 0; i <= numSegments; i++) {
    let t = i / numSegments;
    let point = curve.get(t); // Get point at t value (0 to 1)
    linePoints.push(point);
}

// Convert the points into a new SVG path string using 'M' and 'L' commands
function pointsToLineD(points) {
    let d = `M ${points[0].x} ${points[0].y} `;
    for (let i = 1; i < points.length; i++) {
        d += `L ${points[i].x} ${points[i].y} `;
    }
    return d;
}

let linearPathD = pointsToLineD(linePoints);

// Update an existing path element or create a new one with the line segments
let linearEl = document.getElementById('linearPath');
linearEl.setAttribute('d', linearPathD);
